# Save, apply, rename and delete a layer preset; make a layer from one
> new
> edit Realm
  /realms/west  added realm
> run map.add {"name":"north","to":"/realms/west"}
  /realms/west/north  added map
  /realms/west/north/grid  added grid
  /realms/west/north/paper  added paper
> run layer.add {"name":"hills","to":"/realms/west/north"}

  /realms/west/north/hills  added layer
> run layer.add {"name":"woods","to":"/realms/west/north"}

  /realms/west/north/woods  added layer
> edit Seed
  /realms/west/north/hills  layer.seed  unset → 123
> run preset.save {"at":"/realms/west/north/hills","label":"Mountains"}

> run preset.save {"at":"/realms/west/north/hills","label":"Mountains"}

> presets at /realms/west/north/woods

> run preset.apply {"at":"/realms/west/north/woods","preset":{"user":"Mountains"}}
  /realms/west/north/woods  layer.seed  unset → 123
> current preset at /realms/west/north/woods
